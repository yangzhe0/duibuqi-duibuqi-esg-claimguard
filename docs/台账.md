## 处理链路

1. 使用 CNINFO 搜索接口下载 ESG 相关 PDF，生成 `data/raw_pdfs/`、`data/report_index.csv` 和 `data/download_log.csv`。
2. 对 `data/report_index.csv` 与 `data/raw_pdfs/` 做文件级一致性校验。
3. 使用 MinerU 3.4.0 对 200 份 PDF 批量解析。
4. 保留每份报告的 Markdown、两版结构化 JSON 和图片资源。
5. 汇总生成 `data/parsed_reports_v1/manifest.csv`。
6. 清理解析过程目录，仅保留当前协作基线目录 `data/parsed_reports_v1/`。

```bash

  --target 200 \
  --max-pages 80 \
  --page-size 30 \
  --keyword ESG报告 \
  --out-dir data/raw_pdfs \
  --index data/report_index.csv \
  --log data/download_log.csv
```

校验候选 PDF 和主索引：

整理当前协作基线：

python3 - <<'PY'
from pathlib import Path
import shutil

src_root = Path("data/mineru_outputs/hybrid_high_all")
dst_root = Path("data/parsed_reports_v1/reports")
dst_root.mkdir(parents=True, exist_ok=True)

for report_dir in sorted(p for p in src_root.iterdir() if p.is_dir()):

    if img_src.exists():
        shutil.copytree(img_src, dst / "images", dirs_exist_ok=True)

        (dst / "images").mkdir(exist_ok=True)

生成全量 manifest：

```bash
python3 - <<'PY'
import csv
import re
from pathlib import Path
from pypdf import PdfReader

base = Path(".")
raw_root = base / "data/raw_pdfs"
parsed_root = base / "data/parsed_reports_v1/reports"
out_csv = base / "data/parsed_reports_v1/manifest.csv"

rows = []
for pdf in sorted(raw_root.glob("*.pdf")):
    stem = pdf.stem
    report_dir = parsed_root / stem
    md = report_dir / f"{stem}.md"
    text = md.read_text(encoding="utf-8", errors="ignore")
    lines = text.splitlines()
    image_refs = len(re.findall(r"!\\[[^\\]]*\\]\\([^\\)]+\\)", text))
    image_refs += len(re.findall(r"<img\\b", text, flags=re.I))
    table_markers = sum(1 for line in lines if "|" in line and re.search(r"\\|\\s*[-:]{3,}\\s*\\|", line))
    table_markers += len(re.findall(r"<table\\b", text, flags=re.I))

    parts = stem.split("_")
    rows.append({
        "stock_code": parts[0] if len(parts) > 0 else "",
        "company": parts[1] if len(parts) > 1 else "",
        "report_type": parts[3] if len(parts) > 3 else "",
        "pdf_path": str(pdf),
        "pages": len(PdfReader(str(pdf)).pages),
        "pdf_size_bytes": pdf.stat().st_size,
        "md_path": str(md),
        "md_size_bytes": md.stat().st_size,
        "chars": len(text),
        "lines": len(lines),
        "nonempty_lines": sum(1 for line in lines if line.strip()),
        "headings": sum(1 for line in lines if re.match(r"^#{1,6}\\s+", line)),
        "image_refs": image_refs,
        "table_markers": table_markers,
        "first_nonempty_line": next((line.strip() for line in lines if line.strip()), "")[:180],
        "status": "ok",
    })

with out_csv.open("w", newline="", encoding="utf-8-sig") as fh:
    writer = csv.DictWriter(fh, fieldnames=list(rows[0]))
    writer.writeheader()
    writer.writerows(rows)
PY
```
